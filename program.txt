# The instructions are converted to program.hex and loaded into the
# processor RAM. The "now x is y" are parsed into asserts.vh

# --- BASIC ACCUMULATION & COMBINATIONAL ALU ---
f:7         # now f is 7
f:f+3       # now f is 10
f:f-3       # now f is 7
f:f&4|1^2   # now f is 7

# --- THE MULTI-CYCLE MATH ENGINE ---
a:f*6       #now a is 42 Multiplication
c:a/6+1     #now c is 8 Division (integer)

# --- COMPARISON ---
c<a       # now flag is 1
c>a       # now flag is 0
c=8       # now flag is 1
c=f       # now flag is 0

# --- RADIX FROM REGISTER 'r' ---
r:16        #now r is 16 our radix is now hex.
b:f         #now b is 15 not 7; a-f source digits, not variables
r:a         #now r is 10 decimal; 'a' is 10 in hex, and we were in hex

# --- STACK. POINTER IS 's' ---
s:0         #now s is 0 Clearing the stack
z,42        #now s is 1 push 42 to stack
            # now stack 0 is 42

# --- LOOP ---
s:0         # Clear the stack
l:0         # Initialize a loop counter
[           # Start the loop, push PC
l+1         # Could also be l:l+1
l<3         # Test the exit condition
]           # now l is 3 True? PC:TOS : pop TOS
            # now s is 0 because the stack was emptied

# --- CONDITIONAL ---
a:5
a=5? b:1 ! b:0   # now b is 1 because the condition was true
a<5? c:1 ! c:0   # now c is 0 because a is not less than 5
